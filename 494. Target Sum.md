# Problem No: 494. Target Sum(Medium)

**Link:** [Click here](https://leetcode.com/problems/target-sum/description/)

```cpp
class Solution {
public:
    int findTargetSumWays(vector<int>& nums, int target) {
        int sum = accumulate(nums.begin(), nums.end(), 0);

        if (abs(target) > sum || (target + sum) % 2 != 0 || (sum + target) < 0)
            return 0;
        int p = (target + sum) / 2;
        vector<int> dp(p + 1, 0);
        dp[0] = 1;

        for (auto num : nums) {
            for (int i = p; i >= num; i--) {
                dp[i] += dp[i - num];
            }
        }
        return dp[p];
    }
};
